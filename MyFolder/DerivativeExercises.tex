\begin{problem}
Consider the function $f(x) = 3x^2$.  

~ \\[1em]

\begin{enumerate}
\item Which of the following expressions correctly sets up the limit definition for $f'(x)$?

\begin{multipleChoice}
    \choice{$f'(x) = \lim\limits_{\Delta x \to 0} \frac{3x^2 + \Delta x - 3x^2}{\Delta x}$}
    \choice{$f'(x) = \lim\limits_{\Delta x \to 0} \frac{3(x^2 + \Delta x^2) - 3x^2}{\Delta x}$}
    \choice[correct]{$f'(x) = \lim\limits_{\Delta x \to 0} \frac{3(x + \Delta x)^2 - 3x^2}{\Delta x}$}
    \choice{$f'(x)= \lim\limits_{\Delta x \to 0} \frac{3(x + \Delta x)^2 + 3x^2}{\Delta x}$}
\end{multipleChoice}

\item Expand and simplify the numerator.  

\begin{multipleChoice}
\choice[correct]{$6x \Delta x + (\Delta x)^2$}
\choice{$6  \Delta x + (\Delta x)^2$}
\choice{$6x + (\Delta x)^2$}
\choice{$6x \Delta x + \Delta x$}
\end{multipleChoice}    

\item Complete the problem and state the derivative. 

\[ f'(x) = \answer{6x}  \]

\end{enumerate}
\end{problem}



\begin{problem}
Consider the function $f(x) = 4x^5 - \frac{3}{x^2} + 2\sqrt{x}.$

~ \\[1em]

\begin{enumerate}
\item Rewrite the function with rational and negative exponents.

\[ f(x) = 4x^5 - 3x^{\answer{-2}} + 2x^{\answer{1/2}} \]

\item Compute the derivative using the power rule.

\[ f'(x) = \answer{20x^4 + \frac{6}{x^3} + \frac{1}{\sqrt{x}}}  \]

\end{enumerate}
\end{problem}

\end{problem}


\begin{problem}
Consider the function given below.

\[ y = \frac{\sin(x)}{x^2 + 1} \]


\begin{enumerate}
\item What rule should be used to find the derivative of $y?$


\begin{multipleChoice}
\choice{Power Rule}
\choice{Product Rule}
\choice{Chain Rule}
\choice[correct]{Quotient Rule}
\end{multipleChoice}    


\item Find the derivatives of the numerator and denominator of $y.$

\begin{itemize}
    \item $\frac{d}{dx} (\sin x) = \answer{\cos(x)}$

    \item $\frac{d}{dx} (x^2+1) = \answer{2x}$
\end{itemize}


\item State the derivative of $y.$

\[ \frac{dy}{dx} = \frac{\answer{(x^2+1)\cos(x) - 2x\sin(x)}}{(x^2+1)^2} \]

\end{enumerate}
\end{problem}


\begin{problem}
Find the derivative of $h(x) = \tan^4(x)$.  Remember that this expression can be rewritten as $h(x) = [\tan(x)]^4$.

~ \\[1em]
\begin{enumerate}
\item Apply the chain rule and differentiate the outer power function first.
\[
h'(x) = 4[\tan(x)]^{\answer{3}} \cdot \frac{d}{dx}\left[\tan(x)\right]
\]

\item State the derivative of $\tan(x).$ 

\[ \frac{d}{dx}\left[\tan(x)\right] =  \answer{\sec^2(x)} \]

\item State the simplified form of the derivative of $h.$
\[
h'(x) = \answer{4\tan^3(x)\sec^2(x)}
\]

\end{enumerate}
\end{problem}


\begin{problem}
Follow the steps below to fifferentiate the function.


\[ f(x) = \sqrt{5x + \sin(x)} \]

\begin{enumerate}
\item Treating the square root as an outer power of $1/2$, finish writing the first step of the chain rule.
\[
f'(x) = \frac{1}{2}[5x + \sin(x)]^{1/2} \cdot \frac{d}{dx}\left[ \answer{5x + \sin(x)} \right]
\]

\item Find the derivative of the inner function to complete then the final answer shown below. 
\[
f'(x) = \frac{\answer{5 + \cos(x)}}{2\sqrt{5x + \sin(x)}}
\]

\end{enumerate}
\end{problem}


\begin{problem}
Suppose $f(x)$ is differentiable everywhere.  At a specific point $x = c$, the graph of $f(x)$ is increasing and concave down. 

~ \\[1em]

\begin{enumerate}
\item What can you conclude about the signs of $f'(c)$ and $f''(c)$?

\begin{multipleChoice}
    \choice{$f'(c) > 0$ and $f''(c) > 0$}
    \choice[correct]{$f'(c) > 0$ and $f''(c) < 0$}
    \choice{$f'(c) < 0$ and $f''(c) > 0$}
    \choice{$f'(c) < 0$ and $f''(c) < 0$}
\end{multipleChoice}

~ \\[1em]
\item Which of the following lines could be tangent to $f(x)$ at $x = c$?

\begin{multipleChoice}
    \choice{$y = -3x + 5$}
    \choice{$y = -4$}
    \choice[correct]{$y = 2x - 1$}
\end{multipleChoice}

\begin{hint}
Remember that a function and its tangent line share a point and the \textbf{same} slope at that point.   
\end{hint}

\end{enumerate}
\end{problem}


\begin{problem}
Let $f(x)$ be a function defined on all real numbers.  Identify all statements that are true.


~ \\[1em]

\textbf{Select all that apply.}
\begin{selectAll}
    \choice[correct]{If $f(x)$ is differentiable at $x=c$, then $f(x)$ must be continuous at $x=c$.}
    \choice{If $f(x)$ is continuous at $x=c$, then $f(x)$ must be differentiable at $x=c$.}
    \choice[correct]{If $f(x)$ is not continuous at $x=c$, then $f(x)$ cannot be differentiable at $x=c$.}
    \choice{If $f(x)$ has a sharp corner or cusp at $x=c$, then $\lim_{x \to c} f(x)$ does not exist.}
\end{selectAll}

\end{problem}

\begin{problem}
Let $f(x) = 2\sin(x) + x$ on the closed interval $[0, 2\pi].$  Find all $x$-values where the graph of $f(x)$ has a horizontal 
tangent line.  Recall that a curve has a horizontal tangent line when its derivative is equal to zero. 

~ \\[1em]

\begin{enumerate}
    \item Find the derivative of $f(x).$
\[
f'(x) = \answer{2\cos(x) + 1}
\]

\item Set $f'(x) = 0$ and solve for $\cos(x).$
\[
\cos(x) = \answer{-1/2}
\]

\item Identify the two values of $x$ where this equation is true on the interval $[0, 2\pi].$  
\begin{itemize}
    \item Smaller $x$-value $= \answer{2\pi/3}$
    \item Larger $x$-value $= \answer{4\pi/3}$
\end{itemize}


\item At these points, what is the slope of the normal line (the line perpendicular to the tangent line)?

\begin{multipleChoice}
    \choice{The slope is $0$.}
    \choice{The slope is $-1$.}
    \choice[correct]{The slope is undefined (vertical line).}
    \choice{The slope is $1$.}
\end{multipleChoice}


\end{enumerate}

\end{problem}
