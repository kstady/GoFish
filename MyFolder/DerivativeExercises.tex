\begin{problem}
Let $f(x) = 3x^2$. Which of the following expressions correctly sets up the limit definition for $f'(x)$?

\begin{multipleChoice}
    \choice{$\lim\limits_{\Delta x \to 0} \frac{3x^2 + \Delta x - 3x^2}{\Delta x}$}
    \choice{$\lim\limits_{\Delta x \to 0} \frac{3(x^2 + \Delta x^2) - 3x^2}{\Delta x}$}
    \choice[correct]{$\lim\limits_{\Delta x \to 0} \frac{3(x + \Delta x)^2 - 3x^2}{\Delta x}$}
    \choice{$\lim\limits_{\Delta x \to 0} \frac{3(x + \Delta x)^2 + 3x^2}{\Delta x}$}
\end{multipleChoice}

Expand and simplify the numerator.  

\begin{multipleChoice}
\choice[correct]{$6x \Delta x + (\Delta x)^2$}
\choice{$6  \Delta x + (\Delta x)^2$}
\choice{$6x + (\Delta x)^2$}
\choice{$6x \Delta x + \Delta x$}
\end{multipleChoice}    


Complete the problem and state the derivative. 

\[
f'(x) = \answer{6x}
\]
\end{problem}

\begin{problem}
Consider the function shown below.

\[ y = \frac{\sin(x)}{x^2 + 1} \]

What rule should be used to find the derivative of $y?$

\begin{multipleChoice}
\choice{Power Rule}
\choice{Product Rule}
\choice{Chain Rule}
\choice[correct]{Quotient Rule}
\end{multipleChoice}    


Find the derivatives of the numerator and denominator of $y.$

\begin{itemize}
    \item $\frac{d}{dx} \sin x = \answer{\cos(x)}$

    \item $\frac{d}{dx} (x^2+1) = \answer{2x}$
\end{itemize}

\begin{prompt}
Putting it all together, the final derivative is:
\[
\frac{dy}{dx} = \frac{\answer{(x^2+1)\cos(x) - 2x\sin(x)}}{(x^2+1)^2}
\]
\end{prompt}

\end{problem}


\begin{problem}
Find the derivative of $h(x) = \tan^4(x)$. 
Remember that this expression can be rewritten as $h(x) = (\tan(x))^4$.

Applying the chain rule, we differentiate the outer power function first:
\[
h'(x) = 4(\tan(x))^{\answer{3}} \cdot \frac{d}{dx}\left[\tan(x)\right]
\]
Since the derivative of $\tan(x)$ is $\answer{\sec^2(x)}$, the completely simplified derivative is:
\[
h'(x) = \answer{4\tan^3(x)\sec^2(x)}
\]
\end{problem}


\begin{problem}
Differentiate the function 
\[
F(x) = \sqrt{5x + \sin(x)}
\]
Treating the square root as an outer power of $1/2$, the first step of the chain rule yields:
\[
F'(x) = \frac{1}{2}(5x + \sin(x))^{-1/2} \cdot \frac{d}{dx}\left[ \answer{5x + \sin(x)} \right]
\]
Evaluating the inner derivative gives the final answer:
\[
F'(x) = \frac{\answer{5 + \cos(x)}}{2\sqrt{5x + \sin(x)}}
\]
\end{problem}


\begin{problem}
Suppose $f(x)$ is a differentiable function everywhere. At a specific point $x = c$, the graph of $y = f(x)$ is **increasing** and **concave down**. 

What can you conclude about the signs of $f'(c)$ and $f''(c)$?
\begin{multipleChoice}
    \choice{$f'(c) > 0$ and $f''(c) > 0$}
    \choice[correct]{$f'(c) > 0$ and $f''(c) < 0$}
    \choice{$f'(c) < 0$ and $f''(c) > 0$}
    \choice{$f'(c) < 0$ and $f''(c) < 0$}
\end{multipleChoice}

Which of the following lines could be tangent to $y = f(x)$ at $x = c$?
\begin{multipleChoice}
    \choice{$y = -3x + 5$}
    \choice{$y = -4$}
    \choice[correct]{$y = 2x - 1$}
\end{multipleChoice}
\end{problem}


\begin{problem}
Let $f(x)$ be a function defined on all real numbers. Select **all** statements that are mathematically true.

\begin{selectAll}
    \choice[correct]{If $f(x)$ is differentiable at $x=c$, then $f(x)$ must be continuous at $x=c$.}
    \choice{If $f(x)$ is continuous at $x=c$, then $f(x)$ must be differentiable at $x=c$.}
    \choice[correct]{If $f(x)$ is not continuous at $x=c$, then $f(x)$ cannot be differentiable at $x=c$.}
    \choice{If $f(x)$ has a sharp corner or cusp at $x=c$, then $\lim_{x \to c} f(x)$ does not exist.}
\end{selectAll}

\begin{feedback}
Remember: Differentiability is a stronger condition than continuity. A classic counterexample to the second choice is $f(x) = |x|$ at $x=0$, which is continuous but has a sharp corner (making it non-differentiable).
\end{feedback}
\end{problem}

\begin{problem}
Let $f(x) = 2\sin(x) + x$ on the closed interval $[0, 2\pi]$. We want to find all $x$-values where the graph of $f(x)$ has a horizontal tangent line.

First, recall that a curve has a horizontal tangent line when its derivative is equal to zero. 
Compute the derivative of $f(x)$:
\[
f'(x) = \answer{2\cos(x) + 1}
\]

Now, set $f'(x) = 0$ and solve for $\cos(x)$:
\[
\cos(x) = \answer{-1/2}
\]

On the interval $[0, 2\pi]$, list the two values of $x$ where this equation is true. Enter the smaller value first.
\begin{itemize}
    \item Smaller $x$-value: $x = \answer{2\pi/3}$
    \item Larger $x$-value: $x = \answer{4\pi/3}$
\end{itemize}

\begin{question}
At these points, what is the slope of the normal line (the line perpendicular to the tangent line)?
\begin{multipleChoice}
    \choice{The slope is $0$.}
    \choice{The slope is $-1$.}
    \choice[correct]{The slope is undefined (vertical line).}
    \choice{The slope is $1$.}
\end{multipleChoice}
\end{question}
\end{problem}
