\begin{problem}
A student is asked to find the indefinite integral of the function $f(x) = 6x^5.$  They write the following statement on their exam.

\[
\int 6x^5 = x^6 + C
\]


Identify an errors in their work.

\begin{multipleChoice}
    \choice{They forgot to write the limits of integration on the integral sign.}
    \choice[correct]{They omitted the differential $dx$ on the left-hand side.}
    \choice{The antiderivative should be $\displaystyle \frac{6x^6}{5}$.}
    \choice{The constant of integration $+C$ is not needed for indefinite integrals.}
    \choice{Their work is correct.}
\end{multipleChoice}
\end{problem}


\begin{problem}
Evaluate the indefinite integral using the basic integration rules.


\[
\int (x^4 + e^{7x} - 3) \ dx = \frac{ \answer{x^5} }{ \answer{5} } + \frac{ e^{\answer{7x}} }{\answer{7}} - \answer{3x}  + C
\]

\end{problem}



\begin{problem}
Evaluate the indefinite integral by first simplifying the integrand, then use the power rule.

\[
\int \frac{x^3 - 5x + 6}{x^2} \ dx
\]

\begin{enumerate}
\item Rewrite the integrand in the appropriate form to find the antiderivative. 

\[
\frac{x^3 - 5x + 6}{x^2} = \answer{x} - \frac{\answer{5}}{\answer{x}} + 6 x^{\answer{-2}}
\]

\item Integrate term by term to get the final answer. 
\[
\int \frac{x^3 - 5x + 6}{x^2} \, dx = \frac{\answer{x^2}}{\answer{2}} - 5 \answer{\ln|x|} - \frac{\answer{6}}{\answer{x}} + C
\]

\end{enumerate}
\end{problem}


\begin{problem}
Evaluate the indefinite integral using a substitution. 


\[
\int x^2 e^{x^3} \ dx
\]


\begin{enumerate}
\item Identify $u$ and find the derivative in differential form.

Let $u = \answer{x^3}$

then $du = \answer{3x^2} \ dx.$

\item Identify the correct integral rewritten in terms of $u.$

\begin{multipleChoice}
\choice{$\displaystyle \int 3 e^u \ du$}
\choice{$\displaystyle \int u^2 e^u \ du$}
\choice[correct]{$\displaystyle \frac{1}{3} \int  e^u \ du$}
\choice{$\displaystyle \int e^u \ du$}
\end{multipleChoice}


\item Evaluate.

\[
\int x^2 e^{x^3} \, dx = \frac{ e^{\answer{x^3}} }{ \answer{3} } + C
\]

\end{enumerate}
\end{problem}



\begin{problem}
Evaluate the indefinite integral.
\[
\int \frac{1}{1 + 49x^2} \, dx
\]

\begin{enumerate}
\item Identify $u$ and find the derivative in differential form. 

Let $u = \answer{7x}$ 

then $du = \answer{7} \ dx.$

\item Identify the correct integral rewritten in terms of $u.$
\begin{multipleChoice}
\choice{$\displaystyle \int 7 \cdot \frac{1}{1 + u^2} \ du$}
\choice{$\displaystyle \int \frac{1}{1 + u^2} \ du$}
\choice{$\displaystyle \frac{1}{7} \int \frac{1}{1 + 7u^2} \ du$}
\choice[correct]{$\displaystyle \frac{1}{7} \int  \cdot \frac{1}{1 + u^2} \ du$}
\end{multipleChoice}

\item Evaluate.  
\[
\int \frac{1}{1 + 49x^2} \, dx = \frac{ \answer{1} }{ \answer{7}} \cdot \answer{\arctan(7x)} + C
\]
\end{enumerate}
\end{problem}



\begin{problem}
Follow the steps below to evaluate the indefinite integral.

\[
\int \frac{\sin x + \cos^3 x }{\cos^2 x} \ dx
\]

\begin{enumerate}
\item First, split the numerator of the integrand into two separate terms,

\[
\int \frac{\sin x}{\cos^2 x} + \answer{\cos x}  \ dx
\]

\item The first term can be rewritten as follows.  Simplify this term using standard trigonometric identities.

\[ \frac{\sin x}{\cos x} \cdot \frac{1}{\cos x} = \answer{\tan x} \cdot \answer{\sec x} \]

\item Lastly, rewrite the simplified integral and integrate term-by-term.
\[
\int \answer{\sec x} \cdot \answer{\tan x} + \answer{\cos x} \ dx = \answer{\sec x} + \answer{\sin x} + C
\]

\end{enumerate}
\end{problem}


\begin{problem}
Follow the steps below to evaluate the definite integral.

 \[ \int_{0}^{\pi/4} \sec^2 x \ dx \]

\begin{enumerate}
\item State the antiderivative.  Do not include $C$ since you are evaluating a definite integral. 

    \[ F(x) = \answer{\tan x} \]


\item Find the value of the definite integral.

    \[ \int_{0}^{\pi/4} \sec^2 x \ dx = \answer{1}   \]

\end{enumerate}
\end{problem}


\begin{problem}
A model rocket is launched vertically upward from the ground at time $t = 0$ seconds.  
Its velocity function is given by 

\[
v(t) = 30t^2 + 4t
\]

where $v(t)$ is measured in meters per second.


\begin{enumerate}
\item Find the position function $s(t).$

\[
s(t) = \answer{10t^3} + \answer{2t^2} + C
\]

\item Use the initial condition $s(0) = 0$ to solve for $C.$

\[
C = \answer{0}
\]

\item How far has the rocket traveled after exactly $2$ seconds?

\[
s(2) = \answer{88} \text{ meters}
\]

\end{enumerate}
\end{problem}
